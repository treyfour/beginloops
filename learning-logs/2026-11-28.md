Today's focus: **Hooks — custom React hooks as the client-side half of a cron pipeline**

A cron job produces data. A GET endpoint exposes it. A custom hook *consumes* it. This is the last piece of the pipeline you've been building. Create the hook:

```ts
// hooks/useHeartbeat.ts
"use client";

import { useEffect, useState } from "react";

interface HeartbeatResult {
  status: "ok" | "degraded" | "unknown";
  timestamp: string | null;
  checks: { webhookEndpoint?: "ok" | "unreachable" };
}

export function useHeartbeat(intervalMs = 30_000) {
  const [data, setData] = useState<HeartbeatResult | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function poll() {
      try {
        const res = await fetch("/api/status");
        const json: HeartbeatResult = await res.json();
        setData(json);
      } catch {
        setData({ status: "unknown", timestamp: null, checks: {} });
      } finally {
        setLoading(false);
      }
    }

    poll();
    const id = setInterval(poll, intervalMs);
    return () => clearInterval(id);
  }, [intervalMs]);

  return { data, loading };
}
```

Then use it in `app/page.tsx`:

```tsx
const { data, loading } = useHeartbeat();
// data.status === "ok" | "degraded" | "unknown"
```

**Note:** You'll also need to build `app/api/status/route.ts` for this to have data — that was yesterday's suggestion. Build both together and you'll close the loop: cron runs → heartbeat cron writes → status endpoint exposes → `useHeartbeat` hook reads → component displays.

**Why this matters:** Custom hooks are how React components stay thin. Instead of each component managing its own `setInterval` + `fetch` + cleanup, you encapsulate the whole polling lifecycle once and compose it anywhere. This exact pattern — `useInterval`, `usePolling`, `useLiveData` — appears in every status page, real-time dashboard, and long-polling UI you'll ever build.
