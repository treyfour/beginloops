```ts
// app/api/cron/heartbeat/route.ts
import { NextRequest, NextResponse } from "next/server";

export async function GET(req: NextRequest) {
  const auth = req.headers.get("authorization");
  if (auth !== `Bearer ${process.env.CRON_SECRET}`) {
    return NextResponse.json({ error: "unauthorized" }, { status: 401 });
  }

  const base = process.env.VERCEL_URL
    ? `https://${process.env.VERCEL_URL}`
    : "http://localhost:3000";

  let webhookStatus: "ok" | "unreachable" = "unreachable";
  try {
    const res = await fetch(`${base}/api/webhooks`, {
      method: "POST",
      headers: { "Content-Type": "application/json", "x-github-event": "ping" },
      body: JSON.stringify({ zen: "heartbeat probe" }),
    });
    webhookStatus = res.ok ? "ok" : "unreachable";
  } catch {
    webhookStatus = "unreachable";
  }

  const payload = {
    status: webhookStatus === "ok" ? "ok" : "degraded",
    timestamp: new Date().toISOString(),
    checks: { webhookEndpoint: webhookStatus },
  };

  console.log("[heartbeat]", JSON.stringify(payload));
  return NextResponse.json(payload);
}
```

Add to `.env.local` (and Vercel's Environment Variables dashboard):

```
CRON_SECRET=your-secret-here
```

Generate a strong one with: `openssl rand -hex 32`

**Why this matters:** Cron routes are public `GET` endpoints — without a secret, anyone who discovers the URL can trigger your scheduled job on demand, skewing logs, hammering downstream services, or triggering side effects. Vercel automatically injects `Authorization: Bearer <CRON_SECRET>` when it fires your cron job, so the real scheduler continues to work while external callers get a 401. This is the same pattern you'll see in every production Vercel project that uses crons.
