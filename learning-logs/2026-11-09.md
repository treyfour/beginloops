```ts
// app/api/cron/heartbeat/route.ts
import { NextResponse } from "next/server";

async function probeWithRetry(url: string, attempt = 0): Promise<"ok" | "unreachable"> {
  try {
    const res = await fetch(url, {
      method: "POST",
      headers: { "Content-Type": "application/json", "x-github-event": "ping" },
      body: JSON.stringify({ zen: "heartbeat probe" }),
    });
    return res.ok ? "ok" : "unreachable";
  } catch {
    if (attempt >= 2) return "unreachable";
    await new Promise((r) => setTimeout(r, 500 * 2 ** attempt)); // 500ms → 1s → 2s
    return probeWithRetry(url, attempt + 1);
  }
}

export async function GET() {
  const base = process.env.VERCEL_URL
    ? `https://${process.env.VERCEL_URL}`
    : "http://localhost:3000";

  const webhookStatus = await probeWithRetry(`${base}/api/webhooks`);

  const payload = {
    status: webhookStatus === "ok" ? "ok" : "degraded",
    timestamp: new Date().toISOString(),
    checks: { webhookEndpoint: webhookStatus },
  };

  console.log("[heartbeat]", JSON.stringify(payload));
  return NextResponse.json(payload);
}
```

**Why `500 * 2 ** attempt`?** That's 500ms, then 1000ms, then 2000ms — classic exponential backoff. It spreads retries out so you don't hammer a service that's already under load.

**Why this matters:** A cold-start delay on Vercel serverless functions or a brief network hiccup can cause a single-attempt probe to fail even though the service is healthy. Retrying with backoff avoids alert fatigue from transient noise while still surfacing genuine outages. This pattern — retry N times with exponential delay, give up and report failure — is the standard approach in every cron job and health-check script that calls an external service.
