Today's focus: **Hooks — custom data-fetching hook tied to your real API**

You have a live heartbeat route at `/api/cron/heartbeat` — but `app/page.tsx` still shows the default Next.js template. A custom hook is the idiomatic React way to encapsulate the fetch logic (polling interval, loading state, error state) and reuse it anywhere.

**Create `app/hooks/useHeartbeat.ts`:**

```ts
"use client";

import { useState, useEffect } from "react";

interface HeartbeatStatus {
  ok: boolean;
  timestamp: string | null;
  error: string | null;
}

export function useHeartbeat(intervalMs = 30_000) {
  const [status, setStatus] = useState<HeartbeatStatus>({
    ok: false,
    timestamp: null,
    error: null,
  });

  useEffect(() => {
    async function ping() {
      try {
        const res = await fetch("/api/cron/heartbeat");
        const data = await res.json();
        setStatus({ ok: res.ok, timestamp: data.timestamp ?? new Date().toISOString(), error: null });
      } catch (err) {
        setStatus((prev) => ({ ...prev, ok: false, error: String(err) }));
      }
    }

    ping(); // run immediately on mount
    const id = setInterval(ping, intervalMs);
    return () => clearInterval(id); // cleanup on unmount — this is the key hook discipline
  }, [intervalMs]);

  return status;
}
```

**Then use it in `app/page.tsx`:**

```tsx
"use client";

import { useHeartbeat } from "./hooks/useHeartbeat";

export default function Home() {
  const { ok, timestamp, error } = useHeartbeat();

  return (
    <main style={{ padding: "2rem", fontFamily: "monospace" }}>
      <h1>beginloops</h1>
      <p>Heartbeat: {ok ? "✅ alive" : "❌ down"}</p>
      {timestamp && <p>Last ping: {new Date(timestamp).toLocaleString()}</p>}
      {error && <p style={{ color: "red" }}>Error: {error}</p>}
    </main>
  );
}
```

**Why this matters:** Custom hooks are how you share stateful logic across components without prop-drilling or duplicating `useEffect` calls. The cleanup function (`clearInterval`) inside the `useEffect` return is the most important thing to internalize — skipping it leaks timers and causes stale updates on unmounted components. This same pattern (`useEffect` + interval + cleanup) is how you'd poll any live API: a webhook event feed, a job-queue status, a build pipeline — all of them reduce to this shape.
