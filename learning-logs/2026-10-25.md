Today's focus: **Webhooks — request authentication with HMAC signatures**

Every production webhook handler must verify that the request came from the expected sender, not an attacker who discovered your URL. GitHub (and Stripe, Svix, etc.) sign every request with a shared secret using HMAC-SHA256 and include the signature in a header. Your handler should reject any request whose signature doesn't match.

```ts
// app/api/webhooks/route.ts
import { createHmac, timingSafeEqual } from "crypto";

const WEBHOOK_SECRET = process.env.WEBHOOK_SECRET ?? "";

function verifySignature(body: string, signature: string | null): boolean {
  if (!signature) return false;
  const expected = "sha256=" + createHmac("sha256", WEBHOOK_SECRET)
    .update(body)
    .digest("hex");
  // timingSafeEqual prevents timing attacks that could leak the secret
  try {
    return timingSafeEqual(Buffer.from(signature), Buffer.from(expected));
  } catch {
    return false; // buffers were different lengths
  }
}

export async function POST(req: Request) {
  const body = await req.text(); // read as raw text to preserve exact bytes for hashing
  const sig = req.headers.get("x-hub-signature-256");

  if (!verifySignature(body, sig)) {
    return new Response("Unauthorized", { status: 401 });
  }

  const payload = JSON.parse(body);
  console.log("Verified webhook payload:", payload);
  return new Response("OK", { status: 200 });
}
```

The two critical details:
1. **Read body as `req.text()`** before parsing JSON — hashing a re-serialized object won't match the original signature.
2. **Use `timingSafeEqual`** — a normal `===` comparison leaks timing info that an attacker can use to guess your secret one byte at a time.

Add `WEBHOOK_SECRET=your-secret-here` to `.env.local`, then set the same value in your GitHub webhook settings.
