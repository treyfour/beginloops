Today's focus: **Webhooks — validating incoming requests with a shared secret**

Without signature verification, anyone who discovers your webhook URL can send fake payloads. Real webhook providers sign every request using HMAC-SHA256 with a secret you share at registration time. Verifying that signature is the single most important thing you can add to a webhook handler.

**Add this to `app/api/webhooks/route.ts`:**

```ts
import { createHmac, timingSafeEqual } from "crypto";

function verifySignature(payload: string, signature: string, secret: string): boolean {
  const expected = createHmac("sha256", secret)
    .update(payload, "utf8")
    .digest("hex");
  const expectedBuf = Buffer.from(`sha256=${expected}`);
  const receivedBuf = Buffer.from(signature);
  // timingSafeEqual prevents timing attacks
  return expectedBuf.length === receivedBuf.length &&
    timingSafeEqual(expectedBuf, receivedBuf);
}

export async function POST(req: Request) {
  const secret = process.env.WEBHOOK_SECRET!;
  const sig = req.headers.get("x-hub-signature-256") ?? "";
  const body = await req.text();

  if (!verifySignature(body, sig, secret)) {
    return new Response("Unauthorized", { status: 401 });
  }

  const payload = JSON.parse(body);
  console.log("Verified webhook payload:", payload);
  return Response.json({ received: true });
}
```

Set `WEBHOOK_SECRET` in `.env.local` and in your Vercel environment variables. Use `timingSafeEqual` (not `===`) to prevent timing side-channel attacks — this is the subtlety that trips up most developers the first time.
