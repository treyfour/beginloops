**Add a `GET` handler to `app/api/webhooks/route.ts` that returns stored webhook events, using a module-level array — this completes the webhook lifecycle (receive → store → retrieve) and teaches you the boundary where in-memory state breaks in production.**

**Step 1 — Update `app/api/webhooks/route.ts` to store and expose events:**

```ts
import { NextRequest, NextResponse } from "next/server";

interface WebhookEvent {
  id: string;
  event: string;
  receivedAt: string;
  payload: unknown;
}

// Module-level store — survives across requests within one serverless instance
const events: WebhookEvent[] = [];
const MAX_EVENTS = 20;

export async function POST(req: NextRequest) {
  const event = req.headers.get("x-github-event") ?? "unknown";
  const payload = await req.json();

  const entry: WebhookEvent = {
    id: `${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
    event,
    receivedAt: new Date().toISOString(),
    payload,
  };

  events.unshift(entry);           // newest first
  if (events.length > MAX_EVENTS) events.pop();  // cap the list

  console.log("[webhook]", event, entry.id);
  return NextResponse.json({ ok: true, id: entry.id });
}

export async function GET() {
  return NextResponse.json({
    count: events.length,
    events,
  });
}
```

**Step 3 — Wire it into `page.tsx` so you can see events in the browser:**

```tsx
"use client";
import { useEffect, useState } from "react";

interface WebhookEvent {
  id: string;
  event: string;
  receivedAt: string;
  payload: unknown;
}

export default function Home() {
  const [events, setEvents] = useState<WebhookEvent[]>([]);

  useEffect(() => {
    fetch("/api/webhooks")
      .then((r) => r.json())
      .then((data) => setEvents(data.events ?? []))
      .catch(console.error);
  }, []);

  return (
    <main style={{ padding: "2rem", fontFamily: "monospace" }}>
      <h1>beginloops — webhook log</h1>
      {events.length === 0 ? (
        <p>No events yet. POST to /api/webhooks to send one.</p>
      ) : (
        <ul style={{ listStyle: "none", padding: 0 }}>
          {events.map((e) => (
            <li key={e.id} style={{ marginBottom: "1rem", borderLeft: "3px solid #0070f3", paddingLeft: "1rem" }}>
              <strong>{e.event}</strong> — {new Date(e.receivedAt).toLocaleString()}
              <pre style={{ fontSize: "0.75rem", color: "#666" }}>
                {JSON.stringify(e.payload, null, 2).slice(0, 200)}…
              </pre>
            </li>
          ))}
        </ul>
      )}
    </main>
  );
}
```

**Why this matters:** This is the canonical shape of every webhook dashboard — Stripe, GitHub, Twilio all work this way. Your POST endpoint is the "receiver"; your GET endpoint (backed by a real DB in production) is the "audit log". The module-level array works perfectly in development, but in production each serverless invocation may be a fresh instance — so events from one request may not be visible from another. That gap — from `let events = []` to Vercel KV or a Postgres table — is exactly what you'll hit next, and it's the most important lesson in building reliable event-driven systems.
