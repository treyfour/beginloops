## Suggested next step
Build a **`useHeartbeat` React hook** in `app/hooks/useHeartbeat.ts` that polls your existing heartbeat endpoint and exposes the system status to any component that needs it:

```ts
// app/hooks/useHeartbeat.ts
"use client";
import { useEffect, useState } from "react";

interface HeartbeatPayload {
  status: "ok" | "degraded";
  timestamp: string;
  checks: { webhookEndpoint: "ok" | "unreachable" };
}

export function useHeartbeat(intervalMs = 30_000) {
  const [data, setData] = useState<HeartbeatPayload | null>(null);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    async function poll() {
      try {
        const res = await fetch("/api/cron/heartbeat");
        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        setData(await res.json());
        setError(null);
      } catch (err) {
        setError(err instanceof Error ? err.message : "unknown error");
      }
    }

    poll(); // fire immediately on mount
    const id = setInterval(poll, intervalMs);
    return () => clearInterval(id); // cleanup on unmount
  }, [intervalMs]);

  return { data, error };
}
```

Then use it in `app/page.tsx` to show a live status indicator:

```tsx
import { useHeartbeat } from "./hooks/useHeartbeat";

export default function Home() {
  const { data, error } = useHeartbeat();
  const color = error ? "red" : data?.status === "ok" ? "green" : "yellow";
  return <p style={{ color }}>System: {error ?? data?.status ?? "loading…"}</p>;
}
```

**Why this matters:** This is the canonical pattern for consuming a backend endpoint in a React app — a custom hook encapsulates the fetch + polling logic and makes it reusable across components. It also closes the feedback loop on the heartbeat cron: instead of only being visible in Vercel logs, the system status becomes something you can actually see. Once you add the `CRON_SECRET` guard (yesterday's suggestion), you'll need to pass the token here too — a concrete example of how auth propagates from API to client.
