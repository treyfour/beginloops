Today's focus: **Hooks (Custom React Hooks)**

You already have a backend heartbeat endpoint. Now close the loop on the frontend: a custom hook is the idiomatic React way to encapsulate polling logic so it can be reused anywhere in the app without copy-pasting `useEffect`/`clearInterval` boilerplate.

**Step 1 — Create `hooks/useHeartbeat.ts`:**
```ts
import { useEffect, useState } from "react";

interface HeartbeatPayload {
  status: "ok" | "degraded";
  timestamp: string;
  checks: { webhookEndpoint: "ok" | "unreachable" };
}

export function useHeartbeat(intervalMs = 10_000) {
  const [data, setData] = useState<HeartbeatPayload | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    let cancelled = false;

    async function poll() {
      try {
        const res = await fetch("/api/cron/heartbeat");
        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        const json: HeartbeatPayload = await res.json();
        if (!cancelled) {
          setData(json);
          setError(null);
        }
      } catch (err) {
        if (!cancelled) setError(String(err));
      } finally {
        if (!cancelled) setLoading(false);
      }
    }

    poll(); // run immediately on mount
    const id = setInterval(poll, intervalMs);

    // cleanup: clear the interval AND set cancelled so in-flight fetches
    // don't call setState after the component has unmounted
    return () => {
      cancelled = true;
      clearInterval(id);
    };
  }, [intervalMs]);

  return { data, loading, error };
}
```

**Step 2 — Use it in `app/page.tsx`:**
```tsx
"use client";
import { useHeartbeat } from "@/hooks/useHeartbeat";

export default function Home() {
  const { data, loading, error } = useHeartbeat(10_000);

  return (
    <main style={{ padding: "2rem", fontFamily: "monospace" }}>
      <h1>beginloops</h1>
      <h2>Heartbeat</h2>
      {loading && <p>checking…</p>}
      {error && <p style={{ color: "red" }}>error: {error}</p>}
      {data && (
        <div>
          <p>status: <strong>{data.status}</strong></p>
          <p>webhook: {data.checks.webhookEndpoint}</p>
          <p style={{ color: "gray", fontSize: "0.8rem" }}>
            last checked: {new Date(data.timestamp).toLocaleTimeString()}
          </p>
        </div>
      )}
    </main>
  );
}
```

**Why the `cancelled` flag matters:** `clearInterval` stops future polls, but a fetch already in-flight when the component unmounts will still resolve. Without the flag, calling `setData` on an unmounted component causes a React warning and can trigger subtle bugs in strict-mode double-effects. The flag makes the cleanup truly safe.

**Why `"use client"` is required:** `useEffect` and `useState` are client-only APIs. Any component that imports a custom hook using them must be a Client Component — add the `"use client"` directive at the top of the file.
