**Step 1 — Add `POST` + `GET` to the heartbeat route:**

```ts
// app/api/cron/heartbeat/route.ts
import { NextResponse } from "next/server";

interface HeartbeatResult {
  status: "ok" | "degraded";
  timestamp: string;
  checks: { webhookEndpoint: "ok" | "unreachable" };
}

let lastBeat: HeartbeatResult | null = null; // module-level state

export async function POST() {
  const base = process.env.VERCEL_URL
    ? `https://${process.env.VERCEL_URL}`
    : "http://localhost:3000";

  let webhookStatus: "ok" | "unreachable" = "unreachable";
  try {
    const res = await fetch(`${base}/api/webhooks`, {
      method: "POST",
      headers: { "Content-Type": "application/json", "x-github-event": "ping" },
      body: JSON.stringify({ zen: "heartbeat probe" }),
    });
    webhookStatus = res.ok ? "ok" : "unreachable";
  } catch {
    webhookStatus = "unreachable";
  }

  lastBeat = {
    status: webhookStatus === "ok" ? "ok" : "degraded",
    timestamp: new Date().toISOString(),
    checks: { webhookEndpoint: webhookStatus },
  };

  console.log("[heartbeat]", JSON.stringify(lastBeat));
  return NextResponse.json(lastBeat);
}

export async function GET() {
  if (!lastBeat) {
    return NextResponse.json({ status: "no heartbeat yet" }, { status: 404 });
  }
  return NextResponse.json(lastBeat);
}
```

**Step 3 — Show heartbeat status in `page.tsx`:**

```tsx
"use client";
import { useEffect, useState } from "react";

type Beat = { status: string; timestamp: string; checks: Record<string, string> };

export default function Home() {
  const [beat, setBeat] = useState<Beat | null>(null);

  useEffect(() => {
    fetch("/api/cron/heartbeat").then(r => r.json()).then(setBeat).catch(() => {});
  }, []);

  const secondsAgo = beat
    ? Math.round((Date.now() - new Date(beat.timestamp).getTime()) / 1000)
    : null;

  return (
    <main style={{ padding: "2rem", fontFamily: "monospace" }}>
      <h1>beginloops</h1>
      {beat ? (
        <p>
          Last heartbeat: <strong>{secondsAgo}s ago</strong> — status:{" "}
          <strong style={{ color: beat.status === "ok" ? "green" : "red" }}>
            {beat.status}
          </strong>
        </p>
      ) : (
        <p>No heartbeat recorded yet — trigger GET /api/cron/heartbeat to run it.</p>
      )}
    </main>
  );
}
```

**Why this matters:** Crons are *triggers*, not *stores*. The moment you want the UI to reflect what a cron last did, you hit the fundamental design question: where do you put the result? Module-level variables work in development but evaporate across serverless instances — the real answer is a database or KV store (like Vercel KV / Redis). Understanding that gap is the bridge from "toy cron" to production-ready background jobs.
