Today's focus: **Hooks — connecting your backend cron output to the UI**

The cron at `/api/cron/heartbeat` already runs daily (per `vercel.json`) and returns a JSON status object. Right now nothing in the browser ever sees that data. A React hook that polls it closes that gap and teaches two important patterns at once: writing a custom hook and seeing live cron output reflected in the UI.

**Step 1 — Create `app/hooks/useHeartbeat.ts`:**

```ts
"use client";
import { useState, useEffect } from "react";

interface HeartbeatResult {
  status: "ok" | "degraded" | "unknown";
  timestamp: string;
  checks: Record<string, string>;
}

export function useHeartbeat(intervalMs = 30_000) {
  const [data, setData] = useState<HeartbeatResult | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    let cancelled = false;

    async function poll() {
      try {
        const res = await fetch("/api/cron/heartbeat");
        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        const json = await res.json();
        if (!cancelled) { setData(json); setError(null); }
      } catch (e) {
        if (!cancelled) setError(e instanceof Error ? e.message : "fetch failed");
      } finally {
        if (!cancelled) setLoading(false);
      }
    }

    poll();
    const id = setInterval(poll, intervalMs);
    return () => { cancelled = true; clearInterval(id); };
  }, [intervalMs]);

  return { data, loading, error };
}
```

**Step 2 — Use the hook in `app/page.tsx`** (add `"use client"` at the top, import and call the hook, render the status):

```ts
"use client";
import { useHeartbeat } from "./hooks/useHeartbeat";

export default function Home() {
  const { data, loading, error } = useHeartbeat();

  return (
    <main>
      <h1>beginloops</h1>
      <p>
        Heartbeat:{" "}
        {loading ? "checking…" : error ? `error — ${error}` : data?.status}
      </p>
    </main>
  );
}
```

**Why this matters:** This is the full loop: a cron writes data → an API route exposes it → a React hook polls it → the UI shows it. Once this works you can see in real-time whether your webhook endpoint is reachable, and you'll understand exactly why the next step (a cached `/api/status` route) is worth building — polling the heartbeat directly from the UI runs all the checks on every poll, which is wasteful.
