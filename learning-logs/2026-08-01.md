**Step 1 — Update `app/api/webhooks/route.ts`:**

import { createHmac, timingSafeEqual } from "crypto";
import { NextRequest, NextResponse } from "next/server";

function verifySignature(body: string, signature: string | null, secret: string): boolean {
  if (!signature || !secret) return false;
  const expected = "sha256=" + createHmac("sha256", secret).update(body).digest("hex");
  const a = Buffer.from(signature);
  const b = Buffer.from(expected);
  if (a.length !== b.length) return false;
  return timingSafeEqual(a, b);
}

export async function POST(req: NextRequest) {
  const raw = await req.text(); // read as text to preserve exact bytes for HMAC
  const sig = req.headers.get("x-hub-signature-256");
  const secret = process.env.GITHUB_WEBHOOK_SECRET ?? "";

  if (!verifySignature(raw, sig, secret)) {
    return NextResponse.json({ error: "Unauthorized" }, { status: 401 });

**Why `timingSafeEqual` matters:** A naive `===` string comparison leaks timing information — an attacker can guess the correct signature one byte at a time by measuring response latency. `timingSafeEqual` takes constant time regardless of where the comparison fails, closing that attack vector. This HMAC-SHA256 + timing-safe pattern is identical across GitHub, Stripe, Slack, and Linear — learn it once and it transfers everywhere.
