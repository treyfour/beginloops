Today's focus: **Webhooks — routing incoming events by type**

Right now your webhook route likely handles every incoming payload the same way. Real services (GitHub, Stripe, Twilio) send dozens of different event types to the same URL. The standard pattern is to read the event type from the request, then dispatch to a dedicated handler function. Here's how to wire it up:

```ts
// app/api/webhooks/route.ts
import { NextRequest, NextResponse } from "next/server";

// --- individual event handlers ---
function handlePush(payload: Record<string, unknown>) {
  const branch = (payload.ref as string)?.replace("refs/heads/", "");
  console.log(`[webhook] push to branch: ${branch}`);
}

function handlePullRequest(payload: Record<string, unknown>) {
  const action = payload.action;
  const title = (payload.pull_request as Record<string, unknown>)?.title;
  console.log(`[webhook] pull_request ${action}: "${title}"`);
}

// --- dispatcher ---
export async function POST(req: NextRequest) {
  const event = req.headers.get("x-github-event") ?? "unknown";
  const payload = await req.json();

  switch (event) {
    case "push":
      handlePush(payload);
      break;
    case "pull_request":
      handlePullRequest(payload);
      break;
    default:
      console.log(`[webhook] unhandled event type: ${event}`);
  }

  return NextResponse.json({ received: true, event });
}
```

Test it locally with curl to see the routing in action:

```bash
# Simulate a push event
curl -X POST http://localhost:3000/api/webhooks \
  -H "Content-Type: application/json" \
  -H "x-github-event: push" \
  -d '{"ref":"refs/heads/main","commits":[]}'

# Simulate a pull_request event
curl -X POST http://localhost:3000/api/webhooks \
  -H "Content-Type: application/json" \
  -H "x-github-event: pull_request" \
  -d '{"action":"opened","pull_request":{"title":"My first PR"}}'
```

**Why this matters:** This dispatcher pattern is how every production webhook handler scales. Whether you're processing GitHub events, Stripe payment confirmations, or Twilio SMS replies, the structure is always the same — one route URL, one header that names the event type, one switch that routes it. Once you have this, adding a new event type is just adding a `case` and a handler function.
