Today's focus: **Hooks — turning your cron/heartbeat endpoint into reactive UI state**

You already have the server-side heartbeat at `/api/cron/heartbeat`. The missing piece is a React hook that a component can call to see that status update in real time. Create `app/hooks/useHeartbeat.ts`:

```ts
import { useState, useEffect, useCallback } from "react";

type HeartbeatStatus = "ok" | "degraded" | "loading" | "error";

interface HeartbeatResult {
  status: HeartbeatStatus;
  timestamp: string | null;
  lastChecked: Date | null;
  refresh: () => void;
}

export function useHeartbeat(intervalMs = 30_000): HeartbeatResult {
  const [status, setStatus] = useState<HeartbeatStatus>("loading");
  const [timestamp, setTimestamp] = useState<string | null>(null);
  const [lastChecked, setLastChecked] = useState<Date | null>(null);

  const check = useCallback(async () => {
    try {
      const res = await fetch("/api/cron/heartbeat");
      if (!res.ok) { setStatus("error"); return; }
      const data = await res.json();
      setStatus(data.status === "ok" ? "ok" : "degraded");
      setTimestamp(data.timestamp ?? null);
      setLastChecked(new Date());
    } catch {
      setStatus("error");
    }
  }, []);

  useEffect(() => {
    check();
    const id = setInterval(check, intervalMs);
    return () => clearInterval(id);
  }, [check, intervalMs]);

  return { status, timestamp, lastChecked, refresh: check };
}
```

Then use it in `app/page.tsx`:

```tsx
import { useHeartbeat } from "./hooks/useHeartbeat";

export default function Home() {
  const { status, lastChecked, refresh } = useHeartbeat(10_000);
  return (
    <main>
      <p>System: <strong>{status}</strong></p>
      {lastChecked && <p>Last checked: {lastChecked.toLocaleTimeString()}</p>}
      <button onClick={refresh}>Refresh now</button>
    </main>
  );
}
```

**Why this matters:** Custom hooks are how you encapsulate async + `useEffect` + polling logic so components stay clean. This pattern — fetch on mount, re-fetch on interval, expose a manual refresh — shows up constantly in real dashboards and monitoring UIs.
