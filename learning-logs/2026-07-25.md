```ts
// app/api/webhooks/route.ts
import { NextRequest, NextResponse } from "next/server";

const events: { receivedAt: string; payload: unknown }[] = [];

export async function POST(req: NextRequest) {
  const payload = await req.json();
  events.unshift({ receivedAt: new Date().toISOString(), payload });
  if (events.length > 20) events.pop();
  return NextResponse.json({ ok: true });
}

export async function GET() {
  return NextResponse.json(events);
}
```

**Step 2 — Add `useWebhookEvents` to `app/page.tsx`:**

```tsx
"use client";
import { useEffect, useState } from "react";

type WebhookEvent = { receivedAt: string; payload: unknown };

function useWebhookEvents(intervalMs = 5000) {
  const [events, setEvents] = useState<WebhookEvent[]>([]);

  useEffect(() => {
    const fetchEvents = async () => {
      const res = await fetch("/api/webhooks");
      if (res.ok) setEvents(await res.json());
    };

    fetchEvents(); // fetch immediately on mount
    const id = setInterval(fetchEvents, intervalMs);
    return () => clearInterval(id); // cleanup on unmount
  }, [intervalMs]);

  return events;
}

export default function Home() {
  const events = useWebhookEvents();

  return (
    <main style={{ padding: "2rem", fontFamily: "monospace" }}>
      <h1>Webhook Events</h1>
      {events.length === 0 && <p>No events yet — POST to /api/webhooks to add one.</p>}
      {events.map((e, i) => (
        <details key={i} style={{ marginBottom: "0.5rem" }}>
          <summary>{e.receivedAt}</summary>
          <pre>{JSON.stringify(e.payload, null, 2)}</pre>
        </details>
      ))}
    </main>
  );
}
```

**Why this matters:** `useEffect` with a `setInterval` + cleanup is the canonical React pattern for polling. The `return () => clearInterval(id)` line is what prevents memory leaks when the component unmounts — understanding that cleanup function is one of the most common React interview questions and one of the most common production bugs when it's missing. Once this works, you've closed the loop: webhook arrives → stored in memory → React hook polls → UI updates live.
