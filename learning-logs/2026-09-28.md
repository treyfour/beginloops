This is the payoff for the cron work you've done: the endpoint already returns structured JSON, you just need a hook to fetch it.

**Step 1 — the hook:**

```ts
// app/hooks/useHeartbeat.ts
"use client";

import { useEffect, useState } from "react";

interface HeartbeatResult {
  status: "ok" | "degraded" | null;
  lastChecked: Date | null;
  loading: boolean;
}

export function useHeartbeat(intervalMs = 30_000): HeartbeatResult {
  const [status, setStatus] = useState<"ok" | "degraded" | null>(null);
  const [lastChecked, setLastChecked] = useState<Date | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function poll() {
      try {
        const res = await fetch("/api/cron/heartbeat");
        const data = await res.json();
        setStatus(data.status ?? "degraded");
      } catch {
        setStatus("degraded");
      } finally {
        setLoading(false);
        setLastChecked(new Date());
      }
    }

    poll(); // run immediately on mount
    const id = setInterval(poll, intervalMs);
    return () => clearInterval(id); // cleanup on unmount
  }, [intervalMs]);

  return { status, lastChecked, loading };
}
```

**Step 2 — wire it into `app/page.tsx`:**

```tsx
// app/page.tsx
"use client";

import { useHeartbeat } from "./hooks/useHeartbeat";

export default function Home() {
  const { status, lastChecked, loading } = useHeartbeat();

  return (
    <main>
      <h1>beginloops</h1>
      <div>
        {loading ? (
          <span>checking…</span>
        ) : (
          <span style={{ color: status === "ok" ? "green" : "red" }}>
            {status === "ok" ? "● healthy" : "● degraded"}
          </span>
        )}
        {lastChecked && (
          <small> — last checked {lastChecked.toLocaleTimeString()}</small>
        )}
      </div>
    </main>
  );
}
```

**Why this matters:** The `useEffect` + `setInterval` + `clearInterval` pattern is the canonical React way to poll anything. The cleanup function (`return () => clearInterval(id)`) is the key detail — without it, every time the component remounts it adds another timer until you have a memory leak. This pattern applies to WebSockets, SSE connections, and any subscription — learn it once, use it everywhere.

> **Tip:** The `intervalMs` parameter lets you control poll frequency from the call site (`useHeartbeat(5_000)` for 5 seconds in dev, `useHeartbeat(60_000)` in prod) without touching the hook internals. That's the composability advantage of custom hooks.
