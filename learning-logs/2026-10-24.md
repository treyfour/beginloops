Today's focus: **React Hooks — bridging your server-side heartbeat to the UI**

You've built the heartbeat endpoint; now wire it to the frontend. A custom React hook is the right abstraction: it hides the polling logic (setInterval, fetch, cleanup) and any component just calls `useHeartbeat()` to get the current status.

```ts
// app/hooks/useHeartbeat.ts
"use client";

import { useEffect, useState } from "react";

interface HeartbeatStatus {
  status: "ok" | "degraded" | "unknown";
  timestamp: string | null;
  checks: { webhookEndpoint: "ok" | "unreachable" } | null;
}

export function useHeartbeat(intervalMs = 30_000): HeartbeatStatus {
  const [data, setData] = useState<HeartbeatStatus>({
    status: "unknown",
    timestamp: null,
    checks: null,
  });

  useEffect(() => {
    async function poll() {
      try {
        const res = await fetch("/api/cron/heartbeat");
        if (!res.ok) throw new Error("non-2xx");
        const json = await res.json();
        setData(json);
      } catch {
        setData((prev) => ({ ...prev, status: "degraded" }));
      }
    }

    poll(); // fire immediately on mount
    const id = setInterval(poll, intervalMs);
    return () => clearInterval(id); // cleanup on unmount — this is the key hook invariant
  }, [intervalMs]);

  return data;
}
```

Then use it in any Server or Client component:

```tsx
// app/page.tsx (add "use client" if not already there)
import { useHeartbeat } from "./hooks/useHeartbeat";

export default function Home() {
  const { status, timestamp } = useHeartbeat();
  return <p>System: {status} {timestamp ? `· ${new Date(timestamp).toLocaleTimeString()}` : ""}</p>;
}
```

The `return () => clearInterval(id)` cleanup function is the most important line — without it the hook leaks a timer every time the component re-mounts (navigating away and back, React strict mode's double-mount in dev). This is the pattern that makes custom hooks robust.
