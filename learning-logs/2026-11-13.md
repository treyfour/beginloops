```ts
// In-process cache (resets on cold start — intentional for learning)
let lastFiredAt: string | null = null;

export async function GET() {
  const now = new Date();
  const previousFire = lastFiredAt;
  lastFiredAt = now.toISOString();

  const hoursSinceLast = previousFire
    ? (now.getTime() - new Date(previousFire).getTime()) / 1_000 / 60 / 60
    : null;

  // "stale" means the cron ran, but too much time has passed since the last run
  const isStale = hoursSinceLast !== null && hoursSinceLast > 25;

  const payload = {
    status: isStale ? "stale" : "ok",
    timestamp: now.toISOString(),
    lastFiredAt: previousFire ?? "first run",
    hoursSinceLast: hoursSinceLast?.toFixed(1) ?? null,
  };

  console.log("[heartbeat]", JSON.stringify(payload));
  return NextResponse.json(payload);
}
```

**Why this matters:** A heartbeat that never goes `"stale"` is not really a heartbeat — it is just a health check. The point of a heartbeat is that *silence is the signal*: if the response shows `status: "stale"`, your Vercel cron did not fire on schedule. This is the dead man's switch pattern — the alarm fires not when something bad happens, but when the expected good thing *stops* happening. The `useHeartbeat` hook you wrote yesterday can now surface this visually: a `"stale"` status means your cron is broken, not just the downstream service.
