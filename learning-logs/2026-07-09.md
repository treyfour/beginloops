## Suggested next step
Build a `useHeartbeat` React hook that polls your heartbeat endpoint every 30 seconds and displays a live status badge on the homepage — this is how you close the loop from cron → API → UI.

```ts
// app/hooks/useHeartbeat.ts
"use client";

import { useEffect, useState } from "react";

interface HeartbeatResult {
  status: "ok" | "degraded" | "loading" | "error";
  timestamp: string | null;
  checks: Record<string, string>;
}

export function useHeartbeat(intervalMs = 30_000): HeartbeatResult {
  const [result, setResult] = useState<HeartbeatResult>({
    status: "loading",
    timestamp: null,
    checks: {},
  });

  useEffect(() => {
    async function poll() {
      try {
        const res = await fetch("/api/cron/heartbeat");
        if (!res.ok) throw new Error("non-ok response");
        const data = await res.json();
        setResult({
          status: data.status,
          timestamp: data.timestamp,
          checks: data.checks ?? {},
        });
      } catch {
        setResult((prev) => ({ ...prev, status: "error" }));
      }
    }

    poll(); // fire immediately
    const id = setInterval(poll, intervalMs);
    return () => clearInterval(id); // cleanup on unmount
  }, [intervalMs]);

  return result;
}
```

Then wire it into your homepage:

```tsx
// app/page.tsx
"use client";
import { useHeartbeat } from "./hooks/useHeartbeat";

export default function Home() {
  const { status, timestamp } = useHeartbeat();

  const colors: Record<string, string> = {
    ok: "bg-green-500",
    degraded: "bg-yellow-500",
    error: "bg-red-500",
    loading: "bg-gray-400",
  };

  return (
    <main className="flex min-h-screen flex-col items-center justify-center gap-4 p-8">
      <h1 className="text-2xl font-bold">beginloops</h1>
      <div className="flex items-center gap-2">
        <span className={`h-3 w-3 rounded-full ${colors[status]}`} />
        <span className="text-sm text-gray-600">
          {status === "loading" ? "Checking…" : `System ${status}`}
        </span>
      </div>
      {timestamp && (
        <p className="text-xs text-gray-400">
          Last checked: {new Date(timestamp).toLocaleTimeString()}
        </p>
      )}
    </main>
  );
}
```

**Why this matters:** A React hook is the standard way to wrap async, side-effectful logic (polling, WebSockets, EventSource) into a clean, reusable unit that any component can consume. The `useEffect` + `setInterval` + cleanup pattern here is the exact same shape as every real-time dashboard you'll ever build. And `clearInterval` in the cleanup function is the detail most beginners miss — without it you get a new interval on every re-render and the component leaks. Once you have this working, extending it to poll `/api/webhooks` for event history is a one-line change: same hook, different URL.
