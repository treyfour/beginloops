## Suggested next step
Add a custom React hook `useHeartbeat` that polls your existing `/api/cron/heartbeat` endpoint every 30 seconds and wire it into the homepage — this is what "hooks" means in a Next.js context and it directly connects to the heartbeat endpoint you already built.

Create `app/hooks/useHeartbeat.ts`:

```ts
// app/hooks/useHeartbeat.ts
"use client";

import { useState, useEffect, useCallback } from "react";

interface HeartbeatResult {
  status: "ok" | "degraded" | null;
  timestamp: string | null;
  loading: boolean;
}

export function useHeartbeat(intervalMs = 30_000): HeartbeatResult {
  const [status, setStatus] = useState<"ok" | "degraded" | null>(null);
  const [timestamp, setTimestamp] = useState<string | null>(null);
  const [loading, setLoading] = useState(true);

  const poll = useCallback(async () => {
    try {
      const res = await fetch("/api/cron/heartbeat");
      const data = await res.json();
      setStatus(data.status);
      setTimestamp(data.timestamp);
    } catch {
      setStatus("degraded");
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    poll();                              // fire immediately on mount
    const id = setInterval(poll, intervalMs);
    return () => clearInterval(id);    // clean up on unmount
  }, [poll, intervalMs]);

  return { status, timestamp, loading };
}
```

Then in `app/page.tsx`, replace the default content with:

```tsx
"use client";
import { useHeartbeat } from "./hooks/useHeartbeat";

export default function Home() {
  const { status, timestamp, loading } = useHeartbeat();
  return (
    <main style={{ fontFamily: "monospace", padding: "2rem" }}>
      <h1>beginloops</h1>
      <p>heartbeat: {loading ? "checking…" : status}</p>
      {timestamp && <p>last checked: {new Date(timestamp).toLocaleTimeString()}</p>}
    </main>
  );
}
```

**Why this matters:** `useEffect` + `setInterval` + cleanup is the canonical React polling pattern — you'll use this shape whenever you want a component to stay in sync with live data (status pages, dashboards, chat). The `useCallback` wrapper on `poll` prevents the effect from re-running on every render. Understanding when cleanup runs (the `return () => clearInterval(id)` line) is the most common source of React memory-leak bugs, so seeing it in a small example makes it stick.
