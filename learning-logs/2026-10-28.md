Today's focus: **Heartbeats — building a visual uptime monitor from your live heartbeat data**

Yesterday you built the `useHeartbeat` hook. Now put it to work: a classic uptime-style status page that shows a rolling pulse of your server's health. Here's how to extend the hook and build the page:

**1. Extend `useHeartbeat` to keep a history buffer:**

```ts
// app/hooks/useHeartbeat.ts (updated)
"use client";
import { useEffect, useState } from "react";

type HeartbeatStatus = "ok" | "error";

interface HeartbeatEntry {
  status: HeartbeatStatus;
  timestamp: string;
}

export function useHeartbeat(intervalMs = 30_000, historySize = 20) {
  const [history, setHistory] = useState<HeartbeatEntry[]>([]);

  useEffect(() => {
    async function ping() {
      const entry: HeartbeatEntry = {
        status: "error",
        timestamp: new Date().toISOString(),
      };
      try {
        const res = await fetch("/api/cron/heartbeat");
        const data = await res.json();
        entry.status = data.status === "ok" ? "ok" : "error";
      } catch {}
      setHistory((prev) => [...prev.slice(-(historySize - 1)), entry]);
    }

    ping();
    const id = setInterval(ping, intervalMs);
    return () => clearInterval(id);
  }, [intervalMs, historySize]);

  return history;
}
```

**2. Build the status page:**

```tsx
// app/status/page.tsx
"use client";
import { useHeartbeat } from "../hooks/useHeartbeat";

export default function StatusPage() {
  const history = useHeartbeat(10_000, 20); // ping every 10s, keep 20 entries
  const latest = history.at(-1);

  return (
    <main style={{ padding: "2rem", fontFamily: "monospace" }}>
      <h1>Server Status</h1>
      <p>
        Current:{" "}
        <strong style={{ color: latest?.status === "ok" ? "green" : "red" }}>
          {latest?.status ?? "waiting…"}
        </strong>
      </p>

      <h2>Last {history.length} heartbeats</h2>
      <div style={{ display: "flex", gap: "4px" }}>
        {history.map((entry, i) => (
          <span
            key={i}
            title={new Date(entry.timestamp).toLocaleTimeString()}
            style={{
              width: 12,
              height: 36,
              borderRadius: 4,
              backgroundColor: entry.status === "ok" ? "#22c55e" : "#ef4444",
              display: "inline-block",
            }}
          />
        ))}
      </div>
      <p style={{ fontSize: "0.75rem", color: "#888" }}>
        Each bar = one ping. Hover for timestamp. Pinging every 10s.
      </p>
    </main>
  );
}
```

Navigate to `http://localhost:3000/status` after running `npm run dev` to watch the bars fill in live.

**Why this matters:** This is the visual foundation of every SaaS status page (think status.github.com). You're combining a cron-style server endpoint, a polling hook, and real-time React state into a single cohesive feature — exactly the pattern used in production monitoring dashboards.
