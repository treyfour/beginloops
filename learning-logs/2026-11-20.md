Today's focus: **Hooks — custom React hook for real-time heartbeat status**

Create `app/hooks/useHeartbeat.ts`:

```ts
"use client";

import { useEffect, useState } from "react";

type Status = "ok" | "error" | "loading";

export function useHeartbeat(intervalMs = 30_000) {
  const [status, setStatus] = useState<Status>("loading");
  const [lastChecked, setLastChecked] = useState<Date | null>(null);

  useEffect(() => {
    async function ping() {
      try {
        const res = await fetch("/api/cron/heartbeat");
        setStatus(res.ok ? "ok" : "error");
      } catch {
        setStatus("error");
      } finally {
        setLastChecked(new Date());
      }
    }

    ping();                          // run immediately on mount
    const id = setInterval(ping, intervalMs);
    return () => clearInterval(id); // cleanup on unmount — prevents memory leaks
  }, [intervalMs]);

  return { status, lastChecked };
}
```

Then use it in `app/page.tsx`:

```tsx
import { useHeartbeat } from "./hooks/useHeartbeat";

export default function Home() {
  const { status, lastChecked } = useHeartbeat();

  return (
    <main>
      <p>Heartbeat: {status}</p>
      {lastChecked && <p>Last checked: {lastChecked.toLocaleTimeString()}</p>}
    </main>
  );
}
```

**Why this matters:** Custom hooks are how you encapsulate stateful logic that isn't UI — polling, timers, subscriptions, WebSockets. The `useEffect` + `setInterval` + cleanup pattern is the standard way to do time-based work in React. The `return () => clearInterval(id)` cleanup is critical: without it, every re-render of the parent would start a new timer, and navigating away would leak timers into the background. This is the same pattern used by `useSWR`, React Query, and every real-time dashboard you've ever used.
