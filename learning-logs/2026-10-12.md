Today's focus: **Hooks — the React pattern that makes your UI react to changing data**

Here is the full implementation:

```ts
// app/hooks/useHeartbeat.ts
"use client";
import { useEffect, useState } from "react";

type HeartbeatStatus = {
  status: "ok" | "degraded" | "loading";
  timestamp: string | null;
};

export function useHeartbeat(intervalMs = 30_000): HeartbeatStatus {
  const [state, setState] = useState<HeartbeatStatus>({
    status: "loading",
    timestamp: null,
  });

  useEffect(() => {
    async function ping() {
      try {
        const res = await fetch("/api/cron/heartbeat");
        const data = await res.json();
        setState({ status: data.status, timestamp: data.timestamp });
      } catch {
        setState((s) => ({ ...s, status: "degraded" }));
      }
    }

    ping(); // fire immediately on mount
    const id = setInterval(ping, intervalMs);
    return () => clearInterval(id); // cleanup on unmount
  }, [intervalMs]);

  return state;
}
```

Then wire it into your page:

```tsx
// app/page.tsx
"use client";
import { useHeartbeat } from "./hooks/useHeartbeat";

export default function Home() {
  const { status, timestamp } = useHeartbeat();
  const color = status === "ok" ? "green" : status === "degraded" ? "red" : "gray";

  return (
    <main>
      <h1>beginloops</h1>
      <p style={{ color }}>
        System: {status} {timestamp ? `· last checked ${new Date(timestamp).toLocaleTimeString()}` : ""}
      </p>
    </main>
  );
}
```

**Why this matters:** `useEffect` + `setInterval` is the standard React pattern for polling. The cleanup function (`clearInterval`) is critical — without it, every time the component remounts you add another timer, and they pile up. This hook pattern also composes: you can reuse `useHeartbeat` in any component without duplicating the fetch logic.
