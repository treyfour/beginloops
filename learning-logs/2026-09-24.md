Today's focus: **React hooks — bridging your server-side cron with a client-side UI**

Your heartbeat cron already runs on the server. A custom hook closes the loop by letting the browser periodically ask "is the server alive?" and reflect that in the UI — the same pattern used in health dashboards and DevOps tooling.

Create `app/hooks/useHeartbeat.ts`:

```ts
"use client";
import { useState, useEffect, useRef } from "react";

type HeartbeatState = {
  status: "ok" | "error" | "pending";
  lastPing: string | null;
  error: string | null;
};

export function useHeartbeat(intervalMs = 30_000): HeartbeatState {
  const [state, setState] = useState<HeartbeatState>({
    status: "pending",
    lastPing: null,
    error: null,
  });
  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);

  async function ping() {
    try {
      const res = await fetch("/api/cron/heartbeat");
      if (!res.ok) throw new Error(`HTTP ${res.status}`);
      setState({ status: "ok", lastPing: new Date().toISOString(), error: null });
    } catch (err) {
      setState((prev) => ({ ...prev, status: "error", error: String(err) }));
    }
  }

  useEffect(() => {
    ping(); // run immediately on mount
    timerRef.current = setInterval(ping, intervalMs);
    return () => {
      if (timerRef.current) clearInterval(timerRef.current);
    };
  }, [intervalMs]);

  return state;
}
```

Then in `app/page.tsx`, add:

```tsx
"use client";
import { useHeartbeat } from "./hooks/useHeartbeat";

export default function Home() {
  const { status, lastPing } = useHeartbeat();
  return (
    <main>
      <p>Heartbeat: {status} {lastPing ? `— last ping ${lastPing}` : ""}</p>
    </main>
  );
}
```

> **Why `useRef` for the timer?** `setInterval` returns an ID you need to clear on unmount. Storing it in a ref (not state) means updating it never triggers a re-render — it's a side-effect detail, not UI data.

**Why this matters:** Custom hooks are how React encapsulates *behavior* — not just UI, but async polling, WebSocket connections, local storage sync, and more. The `useEffect` + `useState` + `useRef` triad is the foundation of nearly every non-trivial hook you'll ever write. Once you understand it here with a simple heartbeat ping, you can apply it to anything that involves time, subscriptions, or external data.
