Today's focus: **Hooks — building a custom React hook to surface heartbeat data in the browser**

The app already has a heartbeat endpoint that returns `{ status, timestamp, checks }`. Right now that data only lives in server logs. A custom React hook makes it visible to the user in real time — and teaches you the exact pattern that powers every polling UI you'll ever build.

**Create `app/hooks/useHeartbeat.ts`:**

```ts
"use client";

import { useState, useEffect } from "react";

interface HeartbeatPayload {
  status: "ok" | "degraded";
  timestamp: string;
  checks: { webhookEndpoint: "ok" | "unreachable" };
}

export function useHeartbeat(intervalMs = 30_000) {
  const [data, setData] = useState<HeartbeatPayload | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    async function poll() {
      try {
        const res = await fetch("/api/cron/heartbeat");
        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        setData(await res.json());
        setError(null);
      } catch (e) {
        setError((e as Error).message);
      } finally {
        setLoading(false);
      }
    }

    poll();
    const id = setInterval(poll, intervalMs);
    return () => clearInterval(id);   // cleanup on unmount
  }, [intervalMs]);

  return { data, loading, error };
}
```

**Then use it in `app/page.tsx`:**

```tsx
"use client";
import { useHeartbeat } from "@/app/hooks/useHeartbeat";

export default function Home() {
  const { data, loading, error } = useHeartbeat();

  if (loading) return <p>Checking status…</p>;
  if (error)   return <p style={{ color: "red" }}>Error: {error}</p>;

  return (
    <main>
      <h1>beginloops</h1>
      <p>Status: <strong>{data?.status}</strong></p>
      <p>Webhook endpoint: {data?.checks.webhookEndpoint}</p>
      <p>Last checked: {data?.timestamp}</p>
    </main>
  );
}
```

**Why this matters:** The `useEffect` + `setInterval` + `return () => clearInterval(id)` pattern is one of the most important in React — the cleanup function prevents memory leaks when the component unmounts. Every real-time dashboard, status page, or live-updating UI uses this exact shape. You're not just displaying data; you're learning React's lifecycle in a context that directly connects to the backend work you've already done.
