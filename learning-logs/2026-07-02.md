## Suggested next step
Create a `useHeartbeat` custom React hook in `app/hooks/useHeartbeat.ts` that polls `/api/cron/heartbeat` on a configurable interval and exposes live status to any component.

```ts
// app/hooks/useHeartbeat.ts
"use client";
import { useEffect, useState } from "react";

type HeartbeatStatus = "ok" | "degraded" | "loading" | "error";

export function useHeartbeat(intervalMs = 30_000) {
  const [status, setStatus] = useState<HeartbeatStatus>("loading");
  const [lastChecked, setLastChecked] = useState<string | null>(null);

  useEffect(() => {
    async function check() {
      try {
        const res = await fetch("/api/cron/heartbeat");
        const data = await res.json();
        setStatus(data.status === "ok" ? "ok" : "degraded");
        setLastChecked(data.timestamp);
      } catch {
        setStatus("error");
      }
    }

    check(); // run immediately on mount
    const id = setInterval(check, intervalMs);
    return () => clearInterval(id); // clean up on unmount
  }, [intervalMs]);

  return { status, lastChecked };
}
```

Then use it in `app/page.tsx` to show a live status badge:

```tsx
// app/page.tsx — add near the top of your default export
import { useHeartbeat } from "./hooks/useHeartbeat";

const { status, lastChecked } = useHeartbeat(15_000); // poll every 15s
// render: <span>{status === "ok" ? "🟢 Live" : "🔴 Degraded"}</span>
```

**Why this matters:** Custom hooks are how React logic gets extracted and reused — but they're also the right abstraction for anything that needs `useEffect` cleanup (timers, subscriptions, event listeners). The `clearInterval` return from `useEffect` teaches the cleanup pattern you'll use everywhere: WebSocket connections, `EventSource` listeners for server-sent events, and Intersection/ResizeObservers all follow the same shape. Once this hook exists, you can drop a status indicator into any page without duplicating the polling logic. It also gives you your first real UI connection to the API layer you've been building.
