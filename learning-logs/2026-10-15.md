**Build a `useHeartbeat` custom React hook that polls `/api/cron/heartbeat` on a set interval and exposes live status to any component.**

Today's focus: **Hooks — turning your heartbeat route into a live UI signal**

Create `app/hooks/useHeartbeat.ts`:

```ts
"use client";

import { useEffect, useState, useCallback } from "react";

interface HeartbeatPayload {
  status: "ok" | "degraded";
  timestamp: string;
  checks: { webhookEndpoint: "ok" | "unreachable" };
}

export function useHeartbeat(intervalMs = 30_000) {
  const [data, setData] = useState<HeartbeatPayload | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [loading, setLoading] = useState(true);

  const poll = useCallback(async () => {
    try {
      const res = await fetch("/api/cron/heartbeat");
      if (!res.ok) throw new Error(`HTTP ${res.status}`);
      const json: HeartbeatPayload = await res.json();
      setData(json);
      setError(null);
    } catch (err) {
      setError(err instanceof Error ? err.message : "fetch failed");
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    poll();
    const id = setInterval(poll, intervalMs);
    return () => clearInterval(id);
  }, [poll, intervalMs]);

  return { data, error, loading };
}
```

Then use it in `app/page.tsx`:

```tsx
import { useHeartbeat } from "./hooks/useHeartbeat";

export default function Home() {
  const { data, error, loading } = useHeartbeat(15_000); // poll every 15 s

  if (loading) return <p>Checking system status…</p>;
  if (error)   return <p style={{ color: "red" }}>Error: {error}</p>;

  return (
    <main>
      <h1>System Status</h1>
      <p>Status: <strong>{data?.status}</strong></p>
      <p>Webhook: {data?.checks.webhookEndpoint}</p>
      <p>Last checked: {data?.timestamp}</p>
    </main>
  );
}
```

**Why this matters:** Custom React hooks (`use*` functions) are the standard way to encapsulate side-effects and shared state in React. `useHeartbeat` shows the full pattern: `useState` for reactive data, `useEffect` for the interval lifecycle, and `useCallback` to keep the `effect` dependency stable. Once you have this hook, any component can call `useHeartbeat()` and automatically show live status — the hook manages the polling, cleanup, and error state internally. This is the same pattern behind real-time dashboards, presence indicators, and live feed components.
