Here is `lib/useHeartbeat.ts`:

```ts
"use client";

import { useEffect, useReducer } from "react";

type HeartbeatStatus = "idle" | "loading" | "ok" | "degraded" | "error";

interface HeartbeatState {
  status: HeartbeatStatus;
  timestamp: string | null;
  checks: Record<string, string>;
}

type Action =
  | { type: "fetch" }
  | { type: "success"; payload: { status: string; timestamp: string; checks: Record<string, string> } }
  | { type: "error" };

function reducer(state: HeartbeatState, action: Action): HeartbeatState {
  switch (action.type) {
    case "fetch":
      return { ...state, status: "loading" };
    case "success":
      return {
        status: action.payload.status === "ok" ? "ok" : "degraded",
        timestamp: action.payload.timestamp,
        checks: action.payload.checks,
      };
    case "error":
      return { ...state, status: "error" };
    default:
      return state;
  }
}

const INITIAL_STATE: HeartbeatState = { status: "idle", timestamp: null, checks: {} };

/**
 * Polls /api/cron/heartbeat every `intervalMs` milliseconds.
 * Returns the latest status so any component can show a live health badge.
 */
export function useHeartbeat(intervalMs = 30_000): HeartbeatState {
  const [state, dispatch] = useReducer(reducer, INITIAL_STATE);

  useEffect(() => {
    let cancelled = false;

    async function poll() {
      dispatch({ type: "fetch" });
      try {
        const res = await fetch("/api/cron/heartbeat");
        if (!res.ok) throw new Error(`${res.status}`);
        const data = await res.json();
        if (!cancelled) dispatch({ type: "success", payload: data });
      } catch {
        if (!cancelled) dispatch({ type: "error" });
      }
    }

    poll(); // immediate first fetch
    const id = setInterval(poll, intervalMs);

    return () => {
      cancelled = true;
      clearInterval(id);
    };
  }, [intervalMs]);

  return state;
}
```

**Then use it in `app/page.tsx`:**

```tsx
import { useHeartbeat } from "@/lib/useHeartbeat";

// Inside your Home component:
const { status, timestamp } = useHeartbeat();

// Render a small badge:
<span className={status === "ok" ? "text-green-500" : "text-red-500"}>
  ● {status} {timestamp ? `· ${new Date(timestamp).toLocaleTimeString()}` : ""}
</span>
```

**Three concepts to lock in:**

1. **Why `useReducer` instead of `useState`?** The heartbeat has three interdependent fields (`status`, `timestamp`, `checks`). `useReducer` keeps them in a single state object so they always update atomically — you never render a moment where `status` is "ok" but `timestamp` is stale.

2. **The `cancelled` flag prevents stale setState.** If the component unmounts while a fetch is in-flight, `cancelled = true` stops the dispatch from firing on a dead component — a very common React memory-leak pattern.

3. **The hook is the bridge.** Your backend has two time-aware systems: webhooks (event-triggered) and crons (schedule-triggered). The hook is how the browser joins that conversation — it turns a point-in-time API response into a reactive, auto-updating value. This pattern generalises: swap the URL and you have `useCronStatus`, `useWebhookHealth`, or any other live-polling hook.
