Today's focus: **Webhooks (Signature Verification)**

GitHub signs every webhook delivery with a secret you choose and sends the signature as the `X-Hub-Signature-256` header. Without checking it, your endpoint accepts any POST request that knows the URL — which means anyone can send fake events. Verifying the signature is the single most important hardening step for a public webhook endpoint.

Set the same value in your GitHub repo under **Settings → Webhooks → [your webhook] → Secret**.

import { NextRequest, NextResponse } from "next/server";

// --- signature verification helper ---
async function verifyGitHubSignature(
  req: NextRequest,
  rawBody: string
): Promise<boolean> {
  const secret = process.env.GITHUB_WEBHOOK_SECRET;
  if (!secret) return false; // no secret configured → reject

  const sigHeader = req.headers.get("x-hub-signature-256") ?? "";
  if (!sigHeader.startsWith("sha256=")) return false;

  const encoder = new TextEncoder();
  const key = await crypto.subtle.importKey(
    "raw",
    encoder.encode(secret),
    { name: "HMAC", hash: "SHA-256" },
    false,
    ["sign"]
  );
  const mac = await crypto.subtle.sign("HMAC", key, encoder.encode(rawBody));
  const expected =
    "sha256=" +
    Array.from(new Uint8Array(mac))
      .map((b) => b.toString(16).padStart(2, "0"))
      .join("");

  // Constant-time comparison to prevent timing attacks
  if (expected.length !== sigHeader.length) return false;
  let mismatch = 0;
  for (let i = 0; i < expected.length; i++) {
    mismatch |= expected.charCodeAt(i) ^ sigHeader.charCodeAt(i);
  }
  return mismatch === 0;
}

// --- push event parser (unchanged) ---
interface PushEvent {
  ref: string;
  before: string;
  after: string;
  pusher: { name: string; email: string };
  commits: Array<{
    id: string;
    message: string;
    timestamp: string;
    author: { name: string; username: string };
    added: string[];
    removed: string[];
    modified: string[];
  }>;
  repository: { full_name: string; html_url: string };
}

function parsePushEvent(body: PushEvent) {
  const branch = body.ref.replace("refs/heads/", "");
  const commits = body.commits ?? [];
  return {
    event: "push",
    repo: body.repository.full_name,
    branch,
    pusher: body.pusher.name,
    before: body.before?.slice(0, 7),
    after: body.after?.slice(0, 7),
    commits: commits.map((c) => ({
      sha: c.id.slice(0, 7),
      message: c.message.split("\n")[0],
      author: c.author.username,
      timestamp: c.timestamp,
      files: { added: c.added, removed: c.removed, modified: c.modified },
    })),
  };
}

export async function POST(req: NextRequest) {
  // Read the raw body once — needed for signature verification AND parsing
  const rawBody = await req.text();

  // Verify the signature before touching the payload
  const valid = await verifyGitHubSignature(req, rawBody);
  if (!valid) {
    console.warn("[webhook] rejected request — invalid signature");
    return NextResponse.json({ error: "Forbidden" }, { status: 403 });
  }

  const event = req.headers.get("x-github-event") ?? "unknown";
  const body = JSON.parse(rawBody) as PushEvent;

  if (event === "push") {
    const summary = parsePushEvent(body);
    console.log(`\n── push ─ ${summary.repo} ─ ${summary.branch} ──`);
    console.log(`   by: ${summary.pusher}`);
    console.log(`   ${summary.before} → ${summary.after}`);
    for (const c of summary.commits) {
      console.log(`   ${c.sha} ${c.message}`);
      if (c.files.added.length) console.log(`     + ${c.files.added.join(", ")}`);
      if (c.files.modified.length) console.log(`     ~ ${c.files.modified.join(", ")}`);
      if (c.files.removed.length) console.log(`     - ${c.files.removed.join(", ")}`);
    }
    return NextResponse.json(summary);
  }

  console.log(`\n── ${event} ──`);
  console.log(JSON.stringify(body, null, 2));
  return NextResponse.json({ event, received: true });
}

**Step 3 — Test it locally with `curl`:**
```bash
# Without signature → should return 403
curl -X POST http://localhost:3000/api/webhooks \
  -H "Content-Type: application/json" \
  -H "x-github-event: ping" \
  -d '{"zen":"Practicality beats purity."}'

# With correct signature → should return 200
SECRET=your-generated-secret-here
PAYLOAD='{"zen":"Practicality beats purity."}'
SIG=$(echo -n "$PAYLOAD" | openssl dgst -sha256 -hmac "$SECRET" -hex | awk '{print "sha256="$2}')
curl -X POST http://localhost:3000/api/webhooks \
  -H "Content-Type: application/json" \
  -H "x-github-event: ping" \
  -H "X-Hub-Signature-256: $SIG" \
  -d "$PAYLOAD"
```

**Why `req.text()` instead of `req.json()`:** The HMAC is computed over the raw bytes GitHub sent. If you parse with `JSON.parse()` first and then re-serialize, whitespace differences can make the signature mismatch. Always read `rawBody = await req.text()` and sign that, then `JSON.parse(rawBody)` for the data.

**Why constant-time comparison:** A naïve `===` string comparison short-circuits on the first mismatched character, leaking timing information about how close an attacker's guess is. The `mismatch |= a ^ b` loop always runs the full length regardless of where the mismatch is.
