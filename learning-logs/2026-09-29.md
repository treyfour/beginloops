Today's focus: **Hooks — closing the loop between your cron endpoint and the UI**

You have a running cron that returns JSON health data. A custom hook is the cleanest way to consume it from any component. Here's the minimum viable implementation:

```ts
// app/hooks/useHeartbeat.ts
"use client";
import { useEffect, useState } from "react";

export function useHeartbeat(intervalMs = 30_000) {
  const [status, setStatus] = useState<"ok" | "degraded" | null>(null);
  const [lastChecked, setLastChecked] = useState<Date | null>(null);

  useEffect(() => {
    async function poll() {
      try {
        const res = await fetch("/api/cron/heartbeat");
        const data = await res.json();
        setStatus(data.status ?? "degraded");
      } catch {
        setStatus("degraded");
      }
      setLastChecked(new Date());
    }
    poll();
    const id = setInterval(poll, intervalMs);
    return () => clearInterval(id); // ← the key line: without this you leak timers
  }, [intervalMs]);

  return { status, lastChecked };
}
```

Then in `app/page.tsx`:
```tsx
"use client";
import { useHeartbeat } from "./hooks/useHeartbeat";

export default function Home() {
  const { status, lastChecked } = useHeartbeat();
  return (
    <main>
      <span style={{ color: status === "ok" ? "green" : "red" }}>
        {status ?? "…"}
      </span>
      {lastChecked && <small> checked {lastChecked.toLocaleTimeString()}</small>}
    </main>
  );
}
```

**Why the cleanup matters:** `return () => clearInterval(id)` runs when the component unmounts. Skip it and every remount (hot reload, navigation, StrictMode double-mount) stacks another interval — you end up with 4, 8, 16 simultaneous polls. This pattern applies to `setTimeout`, WebSocket `.close()`, and `EventEmitter.off()` — it is the single most important `useEffect` rule.
