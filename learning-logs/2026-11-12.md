**Create a `useHeartbeat` custom React hook at `app/hooks/useHeartbeat.ts` that polls your heartbeat API client-side and surfaces live status in the UI.**

Today's focus: **Hooks — building a custom React hook that connects your front-end to the backend heartbeat you already built**

```ts
// app/hooks/useHeartbeat.ts
import { useState, useEffect } from "react";

type HeartbeatStatus = "ok" | "degraded" | "loading" | "error";

export function useHeartbeat(intervalMs = 30_000) {
  const [status, setStatus] = useState<HeartbeatStatus>("loading");
  const [lastChecked, setLastChecked] = useState<Date | null>(null);

  useEffect(() => {
    async function check() {
      try {
        const res = await fetch("/api/cron/heartbeat");
        const data = await res.json();
        setStatus(data.status === "ok" ? "ok" : "degraded");
      } catch {
        setStatus("error");
      }
      setLastChecked(new Date());
    }

    check();
    const id = setInterval(check, intervalMs);
    return () => clearInterval(id);
  }, [intervalMs]);

  return { status, lastChecked };
}
```

Then use it in `app/page.tsx`:

```tsx
import { useHeartbeat } from "./hooks/useHeartbeat";

export default function Home() {
  const { status, lastChecked } = useHeartbeat();
  return (
    <main>
      <p>API status: <strong>{status}</strong></p>
      {lastChecked && <p>Last checked: {lastChecked.toLocaleTimeString()}</p>}
    </main>
  );
}
```

**Why this matters:** Custom hooks are how React encapsulates stateful logic that isn't tied to any one component. `useHeartbeat` is a great first hook because it combines three fundamentals — `useState` for reactive data, `useEffect` for side effects, and cleanup via `clearInterval` to avoid memory leaks. The return value from `clearInterval` is the "cleanup function" pattern — this is the hook equivalent of a cron's stop signal. You'll use this same pattern everywhere: polling, subscriptions, WebSocket connections.
